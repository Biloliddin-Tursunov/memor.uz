
-- 1. Profiles jadvali (Foydalanuvchilar)
create table if not exists public.profiles (
  id text not null primary key, -- user id
  full_name text not null,
  bio text,
  avatar_url text,
  role text,
  is_volunteer boolean default false,
  rank_weight integer default 100,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Tasks jadvali (Vazifalar)
create table if not exists public.tasks (
  id text not null primary key,
  team_id text not null,
  creator_id text,
  title text not null,
  description text,
  task_type text,
  priority text,
  status text check (status in ('To Do', 'Doing', 'Done')),
  deadline text,
  resource_link text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Team Participants jadvali (Jamoa a'zolari bog'lanishi)
create table if not exists public.team_participants (
  id bigint generated by default as identity primary key,
  user_id text not null references public.profiles(id) on delete cascade,
  team_id text not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  unique(user_id, team_id)
);

-- 4. Row Level Security (Xavfsizlik) yoqish
alter table public.profiles enable row level security;
alter table public.tasks enable row level security;
alter table public.team_participants enable row level security;

-- 5. Politsiyalar (Hamma o'qiy olishi va yozishi uchun ochiq qilib qo'yildi)
-- Profiles
create policy "Public profiles are viewable by everyone." on public.profiles for select using ( true );
create policy "Enable insert for everyone." on public.profiles for insert with check ( true );
create policy "Enable update for everyone." on public.profiles for update using ( true );
create policy "Enable delete for everyone." on public.profiles for delete using ( true );

-- Tasks
create policy "Public tasks are viewable by everyone." on public.tasks for select using ( true );
create policy "Enable insert for tasks." on public.tasks for insert with check ( true );
create policy "Enable update for tasks." on public.tasks for update using ( true );
create policy "Enable delete for tasks." on public.tasks for delete using ( true );

-- Team Participants
create policy "Public participants are viewable by everyone." on public.team_participants for select using ( true );
create policy "Enable all access for participants." on public.team_participants for all using ( true ) with check ( true );
